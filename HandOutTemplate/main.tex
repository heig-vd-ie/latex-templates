\newtheoremstyle{accentexample}
  {10pt}{10pt}                          % space above/below
  {\normalfont}                         % body font
  {}                                    % indent
  {\bfseries\color{accenthesso}}        % head font — the accent
  {}                                    % punctuation after head
  {\newline}                               % space after head
  {\thmname{#1}\thmnumber{ #2}\thmnote{ -- #3}}
\theoremstyle{accentexample}
\newtheorem{examplein}{Example}[chapter]
% --- Wrapper: bar over the head + reduced left/right margins ---
\newenvironment{example}[1][]{%
  \par\vspace{0pt}%
  \begin{adjustwidth}{2.5em}{2.5em}%        % ← bleeds into both margins
  \noindent{\color{accenthesso!60}\rule{0.35\linewidth}{0.6pt}}\par\nobreak\vspace{-1pt}%
  \begin{examplein}[#1]\ignorespaces
}{%
  \end{examplein}%
  \end{adjustwidth}%
  \vspace{0pt}%
}
